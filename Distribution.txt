% Wheel torques: rear-drive + bounded ay_bar-based brake split (final recommended version)
% Sign conventions:
%   Tt >= 0 : total rear drive torque
%   Tb <= 0 : total brake torque
%
% Design goals satisfied:
%   - exact front/rear brake split: 65% / 35%
%   - exact rear-drive split: 50% / 50%
%   - left/right brake asymmetry allowed in cornering
%   - no ABS logic inside torque law
%   - Kf, Kr are constants (best for NLP), but chosen from geometry and
%     safely bounded using the full ay_bar limits of your NLP

kb = 0.65;   % exact front brake fraction

% ===== Exact front/rear brake totals =====
Tbf = kb * Tb;          % total front brake torque <= 0
Tbr = (1-kb) * Tb;      % total rear  brake torque <= 0

% ===== Full feasible ay_bar range from your existing NLP state bounds =====
% State 7 is ay_bar, and x_min/x_max are scaled by x_s

% ===== Safe maximum left-right brake bias =====
% z = K*ay_bar enters weights (0.5-z) and (0.5+z).
% Using z_bias_max = 0.30 guarantees weights stay in [0.20, 0.80]
% over the full feasible ay_bar range.
z_bias_max = 0.30;

% ===== Geometry-inspired nominal gains (from your wheel-load structure) =====
D    = veh.twF_h * veh.dr + veh.twR_h * (1 - veh.dr);

Fzf0 = veh.m * veh.g * veh.lr / (veh.lf + veh.lr);   % static front axle load
Fzr0 = veh.m * veh.g * veh.lf / (veh.lf + veh.lr);   % static rear  axle load

Kf_geom = veh.m * veh.hc * veh.dr       / (2 * D * Fzf0);
Kr_geom = veh.m * veh.hc * (1 - veh.dr) / (2 * D * Fzr0);

% ===== One common constant scaling for NLP safety =====
% This preserves the front/rear relative magnitude of the nominal gains,
% while guaranteeing |Kf*ay_bar| <= z_bias_max and |Kr*ay_bar| <= z_bias_max
% over the whole feasible ay_bar range.
eps_K = 1e-12;
Kgeom_max = max(abs([Kf_geom, Kr_geom]));
Kmax_allow = z_bias_max / max(ay_bar_abs_max, eps_K);

alpha_lr = min(1.0, Kmax_allow / max(Kgeom_max, eps_K));

Kr = alpha_lr * Kr_geom;
